This script investigates fitting separate coefficients for each sex for each of the predictors (near-equivalent to having a sex-interaction term, rather than just a sex-offset.)

Just looking here at the NHM specimens, rather than the much smaller Oxford dataset.

Posterior summaries

Male

map_df(1:nrow(TableOfModels_NHM_males), function(i){
  
  load(file = paste('../NHM_modelfitsM1_SexSplitmales/', Set_males[i]))
  
  MODEL_FIT %>%
    as_draws_array() %>%
    summarise_draws("mean","median","sd","mad","quantile2",
                    ~quantile2(.x, probs =c(0.025,0.167,0.833, 0.975))) -> Fit_df
  
  Fit_df$Sp = TableOfModels_NHM_males$Sp[i]
  Fit_df$Sex = "male"
  
  return(Fit_df)

Plotting male and female coefficients against each other

SexComparison <- PS_NHM_Split %>%
  filter(variable %in% c("b_Temp", "b_YearRescale", "b_Latitude")) %>%
  select(Sp, Sex, variable, mean, q2.5, q97.5) %>%
  pivot_wider( names_from = Sex,
               values_from = c(mean, q2.5, q97.5),
               names_sep = "_") %>%
  filter(!is.na(mean_male), !is.na(mean_female)) %>%
  mutate(male_confident = sign(q2.5_male) == sign(q97.5_male),
         female_confident = sign(q2.5_female) == sign(q97.5_female),
         difference_type = (sign(mean_male) != sign(mean_female)) &
           (male_confident | female_confident))
